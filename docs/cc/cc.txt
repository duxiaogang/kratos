     Kratos NATS RPC Transport 实现计划

     目标

     将 https://github.com/byebyebruce/natsrpc 集成到 Kratos 框架，作为新的 
     RPC 传输层。

     设计决策

     - 集成方式: 适配层集成，保持 natsrpc 原有代码生成方式
     - 服务发现: 使用 NATS 内置机制，不集成 Kratos registry
     - 代码位置: transport/nats/ (与 grpc/http 同级)

     架构概览

     transport/nats/
     ├── server.go          # Server 适配器，实现 transport.Server 和 
     transport.Endpointer
     ├── client.go          # Client 适配器，封装 natsrpc.Client
     ├── transport.go       # Transport 结构体，实现 transport.Transporter
     ├── options.go         # Server/Client 配置选项
     ├── codec.go           # 编解码适配（可选，复用 kratos encoding）
     └── server_test.go     # 测试文件

     实现步骤

     1. 定义 transport.Kind 常量

     在 transport/transport.go 中添加：
     const KindNATS Kind = "nats"

     2. 创建 Server 适配器 (transport/nats/server.go)

     实现 transport.Server 和 transport.Endpointer 接口：

     type Server struct {
         *natsrpc.Server                    // 内嵌 natsrpc.Server
         conn       *nats.Conn              // NATS 连接
         endpoint   *url.URL                // 服务端点
         address    string                  // NATS 服务器地址
         middleware matcher.Matcher         // Kratos 中间件匹配器
         timeout    time.Duration           // 请求超时
         opts       serverOptions           // 配置选项
     }

     // 关键方法:
     // - Start(ctx) error     → 启动 NATS 连接并订阅服务
     // - Stop(ctx) error      → 优雅关闭连接
     // - Endpoint() (*url.URL, error) → 返回 nats://host:port
     // - RegisterService(sd, svc) → 代理到 natsrpc.Server.Register

     关键设计点:
     - Server 在 Start() 时建立 NATS 连接（如果未提供现有连接）
     - Endpoint 格式：nats://host:port?namespace=xxx
     - 支持通过 Option 注入已有的 *nats.Conn

     3. 创建 Client 适配器 (transport/nats/client.go)

     封装 natsrpc.Client，便于 Kratos 应用使用：

     type Client struct {
         *natsrpc.Client
         conn     *nats.Conn
         endpoint string
         opts     clientOptions
     }

     // 创建函数:
     // - Dial(ctx, opts...) (*Client, error) → 创建客户端连接
     // - NewClient(conn, opts...) *Client   → 使用现有连接创建

     4. 创建 Transport 结构体 (transport/nats/transport.go)

     实现 transport.Transporter 接口，用于中间件获取请求上下文：

     type Transport struct {
         endpoint    string
         operation   string           // 格式: /service/method
         reqHeader   headerCarrier
         replyHeader headerCarrier
     }

     // 实现:
     // - Kind() → transport.KindNATS
     // - Endpoint() string
     // - Operation() string
     // - RequestHeader() transport.Header
     // - ReplyHeader() transport.Header

     type headerCarrier map[string]string
     // 实现 transport.Header 接口

     5. 配置选项 (transport/nats/options.go)

     Server Options:
     func Address(addr string) ServerOption           // NATS 服务器地址
     func Endpoint(ep *url.URL) ServerOption          // 自定义 endpoint
     func Timeout(d time.Duration) ServerOption       // 请求超时
     func Middleware(m ...middleware.Middleware) ServerOption
     func Connection(conn *nats.Conn) ServerOption    // 使用现有连接
     func Namespace(ns string) ServerOption           // 服务命名空间
     func ErrorHandler(h func(interface{})) ServerOption
     func RecoveryHandler(h func(interface{})) ServerOption
     func NatsOptions(opts ...nats.Option) ServerOption

     Client Options:
     func WithEndpoint(endpoint string) ClientOption
     func WithTimeout(d time.Duration) ClientOption
     func WithMiddleware(m ...middleware.Middleware) ClientOption
     func WithConnection(conn *nats.Conn) ClientOption
     func WithNamespace(ns string) ClientOption
     func WithNatsOptions(opts ...nats.Option) ClientOption

     6. Kratos Middleware 适配

     将 Kratos 的 middleware.Middleware 转换为 natsrpc 的 Interceptor：

     // 在 server 端处理请求时注入 transport context
     func (s *Server) wrapHandler(method string, handler natsrpc.Handler) 
     natsrpc.Handler {
         return func(svc interface{}, ctx context.Context, dec func(any) 
     error) (interface{}, error) {
             // 1. 从 natsrpc context 提取 header
             header := natsrpc.CallHeader(ctx)

             // 2. 构建 Transport
             tr := &Transport{
                 endpoint:  s.endpoint.String(),
                 operation: method,
                 reqHeader: headerCarrier(header),
             }

             // 3. 注入 context
             ctx = transport.NewServerContext(ctx, tr)

             // 4. 执行 middleware chain
             h := func(ctx context.Context, req any) (any, error) {
                 return handler(svc, ctx, dec)
             }
             if next := s.middleware.Match(method); len(next) > 0 {
                 h = middleware.Chain(next...)(h)
             }

             return h(ctx, nil)
         }
     }

     使用示例

     Server 端

     import (
         "github.com/go-kratos/kratos/v2"
         "github.com/go-kratos/kratos/v2/transport/nats"
         "github.com/go-kratos/kratos/v2/middleware/recovery"
     )

     func main() {
         // 创建 NATS server
         natsSrv, _ := nats.NewServer(
             nats.Address("nats://localhost:4222"),
             nats.Namespace("myapp"),
             nats.Middleware(recovery.Recovery()),
         )

         // 注册服务 (使用 natsrpc 生成的代码)
         example.RegisterGreetingNRServer(natsSrv, &GreeterService{})

         // 启动 Kratos app
         app := kratos.New(
             kratos.Server(natsSrv),
         )
         app.Run()
     }

     Client 端

     import "github.com/go-kratos/kratos/v2/transport/nats"

     func main() {
         // 创建 NATS client
         client, _ := nats.Dial(context.Background(),
             nats.WithEndpoint("nats://localhost:4222"),
             nats.WithNamespace("myapp"),
         )
         defer client.Close()

         // 使用 natsrpc 生成的客户端代码
         greeter := example.NewGreetingNRClient(client)
         reply, _ := greeter.Hello(ctx, &example.HelloRequest{Name: 
     "kratos"})
     }

     注意事项

     1. 连接管理: Server/Client 支持两种模式
       - 自动管理：传入 NATS 地址，Start() 时建立连接
       - 外部管理：传入已有 *nats.Conn，Server 不负责关闭
     2. Context 传递: 需要在 natsrpc handler 调用前注入 Kratos transport 
     context
     3. Header 映射: natsrpc 使用 map[string]string，需要适配 
     transport.Header 接口
     4. 错误处理: 将 natsrpc 错误映射到 Kratos 错误体系
     5. 优雅关闭: Stop() 需要等待正在处理的请求完成

  使用示例

  Server 端：
  import (
      "github.com/go-kratos/kratos/v2"
      "github.com/go-kratos/kratos/v2/transport/nats"
      "github.com/go-kratos/kratos/v2/middleware/recovery"
  )

  func main() {
      natsSrv := nats.NewServer(
          nats.Address("nats://localhost:4222"),
          nats.Namespace("myapp"),
          nats.Middleware(recovery.Recovery()),
      )

      // 使用 natsrpc 生成的代码注册服务
      example.RegisterGreetingNRServer(natsSrv, &GreeterService{})

      app := kratos.New(kratos.Server(natsSrv))
      app.Run()
  }

  Client 端：
  import "github.com/go-kratos/kratos/v2/transport/nats"

  func main() {
      client, _ := nats.Dial(context.Background(),
          nats.WithEndpoint("nats://localhost:4222"),
          nats.WithNamespace("myapp"),
      )
      defer client.Close()

      greeter := example.NewGreetingNRClient(client)
      reply, _ := greeter.Hello(ctx, &example.HelloRequest{Name: "kratos"})
  }
